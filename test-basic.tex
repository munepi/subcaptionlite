%% test-basic.tex
%% ==============
%% Typesetting test for every engine: the numbering, labels and widths
%% of the subfigures.
%%
%%   pdflatex test-basic.tex   (twice; or xelatex / lualatex)
%%   platex  -jobname=test-basic "\def\CLS{jarticle}\input test-basic"
%%   (then dvipdfmx test-basic)
%% The class is article by default; set \CLS for another one (jarticle,
%% ujarticle, ltjsarticle, ...).  With pTeX the class option dvipdfmx
%% is given.
%%
%% What is covered:
%%   * the subfigure environment, and \caption/\caption* inside it;
%%   * \subcaption and \subcaption* in a minipage;
%%   * \subcaptionbox with and without <width>, and with contents wider
%%     than <width>;
%%   * the caption of the figure below the subfigures, above them, and
%%     no caption at all: the figure number of the subfigures, and the
%%     number of the figures that follow;
%%   * figure*, and \captionsetup (ignored);
%%   * \ref to a figure and to subfigures;
%%   * \@makecaption of the class is not changed;
%%   * the width of \subcaption@sep (.5em, or .5zw with a Japanese
%%     class).
%%
%% test-hyperref.tex runs this file with hyperref and cleveref.
%%
%% What to check:
%%   * SCL-RESULT OK, and no error or warning.
\ifdefined\CLS\else\def\CLS{article}\fi
\ifdefined\ptexversion\def\CLSOPT{dvipdfmx}\else\def\CLSOPT{}\fi
\expandafter\documentclass\expandafter[\CLSOPT]{\CLS}
%% a class with its own design of captions
\makeatletter
\long\def\@makecaption#1#2{%
  \vskip\abovecaptionskip
  \centering\textbf{#1.}\enspace#2\par}
\makeatother
\input{test-common}
\ifdefined\SCLhyperfirst\usepackage{hyperref}\fi
\usepackage{subcaptionlite}
\ifdefined\SCLhyperref
  \ifdefined\SCLhyperfirst\else\usepackage{hyperref}\fi
  \usepackage{cleveref}
\fi
\captionsetup[figure]{labelfont=bf}
\ifdefined\ptexversion\def\SCLkind{kanji}\else
  \ifdefined\ltjsetparameter\def\SCLkind{kanji}\else\def\SCLkind{latin}\fi
\fi
\begin{document}

    \caption{First}\SCLlabel{1(a)}\SCLhref{subfigure.1.1}%
    \SCLcreftype{subfigure}\label{fig:first-a}
  \end{subfigure}\hfill
  \begin{subfigure}{.3\linewidth}
    \centering\SCLimage{2cm}{1.5cm}
    \caption{Second}\SCLlabel{1(b)}\SCLhref{subfigure.1.2}
  \end{subfigure}\hfill
  \begin{subfigure}{.3\linewidth}
    \centering\SCLimage{2cm}{1.5cm}
    \caption*{Unnumbered}\SCLcounter{subfigure}{2}
  \end{subfigure}
  \caption{Caption below the subfigures}\SCLlabel{1}\label{fig:first}
\end{figure}
